"""Food recognition service: 24-class MobileNetV2 + non-food rejection. TensorFlow is imported lazily, so the API can start
(and /health works) even when the model files are missing. Inference is kept separate from the HTTP routes."""
from __future__ import annotations

import hashlib
import io
import json
import logging
import os
import threading
from pathlib import Path

import numpy as np
from PIL import Image, ImageOps, UnidentifiedImageError

from app.ml.rejection import Thresholds, decide_batch, min_mahalanobis

log = logging.getLogger("smartplate.recognizer")
BACKEND_DIR = Path(__file__).resolve().parents[2]
DEFAULT_MODEL_DIR = BACKEND_DIR / "models" / "production" / "food_classifier"

ALLOWED_FORMATS = {"JPEG", "PNG", "WEBP", "BMP"}
MAX_PIXELS = 40_000_000
MIN_SIDE = 32
# ImageNet-1k classes 923 (plate) .. 957 (pomegranate) and 959 (carbonara) .. 965 (burrito); drinks, "hay" and "cup" excluded.
IMAGENET_FOOD_INDICES = list(range(923, 958)) + list(range(959, 966))
IMAGENET_SANITY = {933: "cheeseburger", 949: "strawberry", 954: "banana", 963: "pizza"}

MSG_NON_FOOD = "This does not appear to be a food image. Please upload a clear photo of food to continue."
MSG_LOW_CONF = "I couldn't identify this food with enough confidence. Please upload a clearer photo of a single food item."
MSG_UNSUPPORTED = "This looks like food, but it is not one of the 24 foods SmartPlateAI can currently recognise."


class ImageError(Exception):
    def __init__(self, code: str, message: str, status: int) -> None:
        super().__init__(message)
        self.code, self.message, self.status = code, message, status


def sha256_file(p: Path) -> str:
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def decode_upload(data: bytes) -> Image.Image:
    """Validate and decode uploaded bytes into an RGB PIL image. Never trusts the filename or declared content type."""
    bad = ImageError("INVALID_IMAGE", "The file could not be read as an image. Please upload a valid JPG, PNG or WebP photo.", 400)
    try:
        probe = Image.open(io.BytesIO(data))
        fmt, (w, h) = probe.format, probe.size
    except (UnidentifiedImageError, OSError, ValueError, Image.DecompressionBombError):
        raise bad from None
    if fmt not in ALLOWED_FORMATS:
        raise ImageError("UNSUPPORTED_MEDIA_TYPE", "This image format is not supported. Please upload a JPG, PNG or WebP photo.", 415)
    if w * h > MAX_PIXELS:
        raise ImageError("IMAGE_TOO_LARGE", "The image has too many pixels. Please upload a smaller photo.", 413)
    if min(w, h) < MIN_SIDE:
        raise ImageError("INVALID_IMAGE", "The image is too small to analyse. Please upload a larger photo.", 400)
    try:
        probe.verify()
        img = Image.open(io.BytesIO(data))
        img = ImageOps.exif_transpose(img).convert("RGB")  # phone photos: honour EXIF rotation; drop alpha like training did
        img.load()
    except (UnidentifiedImageError, OSError, ValueError, SyntaxError, Image.DecompressionBombError):
        raise bad from None
    return img


class FoodRecognizer:
    def __init__(self, model_dir: str | Path | None = None, allow_uncalibrated: bool = False) -> None:
        self.model_dir = Path(model_dir or os.environ.get("FOOD_MODEL_DIR") or DEFAULT_MODEL_DIR)
        self.allow_uncalibrated = allow_uncalibrated or os.environ.get("ALLOW_UNCALIBRATED_REJECTION", "").lower() == "true"
        self.status, self.status_message = "not_loaded", "Model not loaded yet."
        self.class_names: list[str] = []
        self.size = 224
        self.thresholds: Thresholds | None = None
        self.rejection_cfg: dict | None = None
        self._means = self._precision = None
        self._tf = self._multi = self._imagenet = None
        self._load_lock, self._infer_lock = threading.Lock(), threading.Lock()

    # ------------------------------------------------------------------ loading
    @property
    def ready(self) -> bool:
        return self.status == "ready"

    def load(self) -> bool:
        with self._load_lock:
            if self.ready:
                return True
            model_path, names_path = self.model_dir / "best.keras", self.model_dir / "class_names.json"
            if not (model_path.is_file() and names_path.is_file()):
                self.status, self.status_message = "model_missing", (
                    "The food recognition model is not installed. Train it, then copy the chosen run folder to models\\production\\food_classifier.")
                return False
            try:
                import tensorflow as tf  # noqa: PLC0415
                self._tf = tf
                meta = json.loads(names_path.read_text(encoding="utf-8"))
                self.class_names, self.size = list(meta["class_names"]), int(meta["input"]["shape"][0])
                model = tf.keras.models.load_model(str(model_path), compile=False)
                gap = next(layer for layer in model.layers if isinstance(layer, tf.keras.layers.GlobalAveragePooling2D))
                self._multi = tf.keras.Model(model.inputs, [gap.output, model.output])
                if int(model.outputs[0].shape[-1]) != len(self.class_names):
                    raise ValueError("model output size does not match class_names.json")
                if not self._load_rejection(model_path):
                    return False
                self._load_imagenet()
            except Exception:  # noqa: BLE001
                log.exception("food model failed to load")
                self.status, self.status_message = "error", "The food model failed to load. See the server log for details."
                return False
            self.status, self.status_message = "ready", "Food model ready."
            log.info("food model ready (%d classes, input %d, rejection calibrated=%s)", len(self.class_names), self.size, self.thresholds is not None)
            return True

    def _load_rejection(self, model_path: Path) -> bool:
        cfg_path, npz_path = self.model_dir / "rejection_config.json", self.model_dir / "rejection_stats.npz"
        if cfg_path.is_file() and npz_path.is_file():
            cfg = json.loads(cfg_path.read_text(encoding="utf-8"))
            if cfg.get("model_sha256") != sha256_file(model_path):
                self.status, self.status_message = "error", "The rejection calibration does not match this model file. Re-run calibrate_rejection.py."
                return False
            stats = np.load(npz_path)
            self._means, self._precision = stats["means"], stats["precision"]
            self.thresholds, self.rejection_cfg = Thresholds(**cfg["thresholds"]), cfg
            return True
        if self.allow_uncalibrated:
            log.warning("non-food rejection is NOT calibrated; accepting every readable image (development mode)")
            return True
        self.status, self.status_message = "rejection_uncalibrated", (
            "Non-food rejection has not been calibrated for this model, so predictions are disabled. Run scripts\\calibrate_rejection.py.")
        return False

    def _load_imagenet(self) -> None:
        tf = self._tf
        self._imagenet = tf.keras.applications.MobileNetV2(weights="imagenet", include_top=True, input_shape=(224, 224, 3))
        decode = tf.keras.applications.mobilenet_v2.decode_predictions
        for idx, name in IMAGENET_SANITY.items():
            row = np.zeros((1, 1000), np.float32)
            row[0, idx] = 1.0
            if decode(row, top=1)[0][0][1] != name:
                raise ValueError(f"ImageNet class index {idx} is not '{name}'; the food-class index list is wrong")

    # ------------------------------------------------------------------ inference
    def prepare(self, img: Image.Image) -> np.ndarray:
        """Same resize as training: bilinear, antialiased, aspect ratio not preserved. Returns float32 0..255."""
        arr = np.asarray(img, dtype=np.uint8)
        return self._tf.image.resize(arr, [self.size, self.size], method="bilinear", antialias=True).numpy().astype(np.float32)

    def embed(self, x: np.ndarray):
        """x: (N, S, S, 3) float 0..255 -> (24-class probs, 1280-d embeddings, ImageNet food mass)."""
        tf = self._tf
        with self._infer_lock:
            feats, probs = self._multi(x, training=False)
            x224 = x if x.shape[1] == 224 else tf.image.resize(x, [224, 224], method="bilinear", antialias=True).numpy()
            inet = self._imagenet(tf.keras.applications.mobilenet_v2.preprocess_input(np.array(x224, np.float32)), training=False)
        food_mass = inet.numpy()[:, IMAGENET_FOOD_INDICES].sum(axis=1)
        return probs.numpy(), feats.numpy(), food_mass

    def analyze_image(self, img: Image.Image) -> dict:
        probs, feats, fm = self.embed(self.prepare(img)[None])
        p = probs[0]
        top = np.argsort(-p)[:3]
        d = None
        if self.thresholds is not None:
            d = float(min_mahalanobis(feats, self._means, self._precision)[0])
            decision = str(decide_batch(probs, np.array([d]), fm, self.thresholds)[0])
        else:
            decision = "accept"
        return {"decision": decision, "calibrated": self.thresholds is not None, "confidence": float(p[top[0]]),
                "margin": float(p[top[0]] - p[top[1]]),
                "top_k": [{"class_id": int(i), "food": self.class_names[int(i)], "score": round(float(p[i]), 4)} for i in top],
                "signals": {"feature_distance": d, "imagenet_food_mass": float(fm[0])}}

    def info(self) -> dict:
        cfg = self.rejection_cfg or {}
        return {"status": self.status, "message": self.status_message, "n_classes": len(self.class_names), "input_size": self.size,
                "rejection_calibrated": self.thresholds is not None, "calibrated_with_non_food_examples": bool(cfg.get("calibrated_with_negatives")),
                "source_run": cfg.get("source_run"), "thresholds": cfg.get("thresholds")}

========== backend\app\ml\rejection.py ==========

"""Non-food / low-confidence rejection logic. Pure numpy (no TensorFlow), so it is unit-testable.

Three independent signals are combined:
  1. feature distance: Mahalanobis distance of the image embedding to the nearest of the 24 food classes
  2. ImageNet food mass: probability mass an ORIGINAL ImageNet MobileNetV2 puts on food-related ImageNet classes
  3. the 24-class softmax confidence and top1-top2 margin
Softmax confidence alone is never used to decide "this is food".
"""
from __future__ import annotations

from dataclasses import dataclass

import numpy as np


@dataclass
class Thresholds:
    food_mass_low: float   # ImageNet food mass below this = "does not look like food" evidence
    maha_high: float       # feature distance above this = "far from all 24 foods" evidence
    conf_min: float        # minimum top-1 softmax score to name a food
    margin_min: float      # minimum gap between top-1 and top-2 scores


def fit_feature_stats(feats: np.ndarray, labels: np.ndarray, num_classes: int, shrinkage: float = 0.05):
    """Class means + one shared (shrinkage-regularised) precision matrix."""
    feats = np.asarray(feats, np.float64)
    labels = np.asarray(labels)
    if (np.bincount(labels, minlength=num_classes) == 0).any():
        raise ValueError("every class needs at least one training embedding")
    d = feats.shape[1]
    means = np.stack([feats[labels == c].mean(axis=0) for c in range(num_classes)])
    centered = feats - means[labels]
    cov = centered.T @ centered / max(len(feats) - 1, 1)
    cov = (1.0 - shrinkage) * cov + shrinkage * (np.trace(cov) / d) * np.eye(d)
    return means.astype(np.float32), np.linalg.inv(cov).astype(np.float32)


def min_mahalanobis(feats: np.ndarray, means: np.ndarray, precision: np.ndarray) -> np.ndarray:
    """Squared Mahalanobis distance to the nearest class mean, for a batch of embeddings."""
    f = np.asarray(feats, np.float64)
    m = np.asarray(means, np.float64)
    p = np.asarray(precision, np.float64)
    fp = f @ p
    d = np.einsum("nd,nd->n", fp, f)[:, None] - 2.0 * (fp @ m.T) + np.einsum("kd,kd->k", m @ p, m)[None, :]
    return np.maximum(d.min(axis=1), 0.0)


def decide_batch(probs: np.ndarray, d_min: np.ndarray, food_mass: np.ndarray, th: Thresholds) -> np.ndarray:
    probs = np.asarray(probs)
    srt = np.sort(probs, axis=1)
    conf, margin = srt[:, -1], srt[:, -1] - srt[:, -2]
    out = np.full(len(probs), "accept", dtype=object)
    far = np.asarray(d_min) > th.maha_high
    non_food = far & (np.asarray(food_mass) < th.food_mass_low)
    out[~far & ((conf < th.conf_min) | (margin < th.margin_min))] = "uncertain"
    out[far & ~non_food] = "unsupported_food"   # looks food-like to ImageNet but far from the 24 supported foods
    out[non_food] = "non_food"                  # both independent signals say "not food"
    return out


def wilson_interval(k: int, n: int, z: float = 1.96) -> tuple[float, float]:
    if n == 0:
        return (float("nan"), float("nan"))
    p = k / n
    den = 1 + z * z / n
    centre = (p + z * z / (2 * n)) / den
    half = z * np.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / den
    return (float(max(0.0, centre - half)), float(min(1.0, centre + half)))


def choose_gate_thresholds(pos_fm, pos_md, neg_fm=None, neg_md=None, max_false_reject: float = 0.02) -> dict:
    """Pick (food_mass_low, maha_high) from the calibration half only.
    Without negatives: conservative percentiles of the valid-food data (P2 of food mass, P99 of distance).
    With negatives: maximise non-food rejection subject to falsely rejecting at most `max_false_reject` of valid foods."""
    pos_fm, pos_md = np.asarray(pos_fm), np.asarray(pos_md)
    t_md_grid = np.quantile(pos_md, [0.90, 0.95, 0.97, 0.98, 0.99, 0.995, 0.999])
    t_fm_grid = np.quantile(pos_fm, [0.005, 0.01, 0.02, 0.05, 0.10, 0.20, 0.30, 0.50])
    have_neg = neg_fm is not None and len(neg_fm) >= 30
    best = None
    for t_md in t_md_grid:
        for t_fm in t_fm_grid:
            false = float(np.mean((pos_fm < t_fm) & (pos_md > t_md)))
            if false > max_false_reject:
                continue
            rej = float(np.mean((np.asarray(neg_fm) < t_fm) & (np.asarray(neg_md) > t_md))) if have_neg else 0.0
            key = (rej, -false)
            if best is None or key > best[0]:
                best = (key, float(t_fm), float(t_md), false, rej)
    if not have_neg or best is None:
        t_md, t_fm = float(np.quantile(pos_md, 0.99)), float(np.quantile(pos_fm, 0.02))
        return {"food_mass_low": t_fm, "maha_high": t_md, "method": "positive-only percentiles (P2 food mass, P99 distance)",
                "false_reject_calib": float(np.mean((pos_fm < t_fm) & (pos_md > t_md))), "neg_reject_calib": None}
    return {"food_mass_low": best[1], "maha_high": best[2], "method": f"grid search on calibration negatives (false-reject <= {max_false_reject:.0%})",
            "false_reject_calib": best[3], "neg_reject_calib": best[4]}


def choose_conf_margin(conf, margin, correct, target_acc: float = 0.95, min_accepted: int = 50) -> dict:
    """Risk-coverage calibration on validation predictions: keep the most predictions whose accuracy still meets the target."""
    conf, margin, correct = np.asarray(conf), np.asarray(margin), np.asarray(correct, bool)
    best, fallback = None, None
    for tc in np.concatenate([[0.0], np.quantile(conf, np.arange(0.02, 0.92, 0.02))]):
        for tm in np.concatenate([[0.0], np.quantile(margin, np.arange(0.05, 0.60, 0.05))]):
            acc_mask = (conf >= tc) & (margin >= tm)
            n = int(acc_mask.sum())
            if n < min_accepted:
                continue
            acc, cov = float(correct[acc_mask].mean()), n / len(conf)
            if acc >= target_acc and (best is None or (cov, acc) > (best[2], best[1])):
                best = (float(tc), acc, cov, float(tm))
            if fallback is None or acc > fallback[1]:
                fallback = (float(tc), acc, cov, float(tm))
    chosen, met = (best, True) if best else (fallback, False)
    if chosen is None:
        return {"conf_min": 0.0, "margin_min": 0.0, "accuracy_of_accepted": None, "coverage": None, "target_met": False}
    return {"conf_min": chosen[0], "margin_min": chosen[3], "accuracy_of_accepted": chosen[1], "coverage": chosen[2], "target_met": met}

"""POST /api/analyze/food and GET /api/models/status."""
from __future__ import annotations

import logging
import os
import threading
import time

from fastapi import APIRouter, Depends, File, Form, UploadFile
from fastapi.responses import JSONResponse

from app.ml.recognizer import MSG_LOW_CONF, MSG_NON_FOOD, MSG_UNSUPPORTED, FoodRecognizer, ImageError, decode_upload
from app.services.nutrition import NutritionReference

log = logging.getLogger("smartplate.api")
router = APIRouter(prefix="/api", tags=["analysis"])
MAX_MB = float(os.environ.get("MAX_UPLOAD_MB", "8"))
MAX_BYTES = int(MAX_MB * 1024 * 1024)

_recognizer: FoodRecognizer | None = None
_nutrition: NutritionReference | None = None
_singleton_lock = threading.Lock()


def get_recognizer() -> FoodRecognizer:
    global _recognizer
    with _singleton_lock:
        if _recognizer is None:
            _recognizer = FoodRecognizer()
    return _recognizer


def get_nutrition() -> NutritionReference:
    global _nutrition
    with _singleton_lock:
        if _nutrition is None:
            _nutrition = NutritionReference()
    return _nutrition


def error(code: str, message: str, status: int, **extra) -> JSONResponse:
    return JSONResponse(status_code=status, content={"success": False, "error_code": code, "message": message, **extra})


@router.get("/models/status")
def models_status(rec: FoodRecognizer = Depends(get_recognizer)) -> dict:
    rec.load()
    return {"food_classifier": rec.info()}


@router.post("/analyze/food")
def analyze_food(file: UploadFile = File(...), serving_grams: float | None = Form(None),
                 rec: FoodRecognizer = Depends(get_recognizer), nutrition: NutritionReference = Depends(get_nutrition)):
    if serving_grams is not None and not (1 <= serving_grams <= 2000):
        return error("INVALID_SERVING", "Serving size must be between 1 and 2000 grams.", 422)
    data = file.file.read(MAX_BYTES + 1)
    if len(data) > MAX_BYTES:
        return error("FILE_TOO_LARGE", f"The image is larger than {MAX_MB:g} MB. Please upload a smaller photo.", 413)
    if not data:
        return error("INVALID_IMAGE", "The uploaded file is empty. Please choose a photo and try again.", 400)
    if not rec.load():
        return error("MODEL_UNAVAILABLE", rec.status_message, 503)
    t0 = time.time()
    try:
        img = decode_upload(data)
        result = rec.analyze_image(img)
    except ImageError as exc:
        return error(exc.code, exc.message, exc.status)
    except Exception:  # noqa: BLE001
        log.exception("analysis failed")
        return error("INTERNAL_ERROR", "Something went wrong while analysing the image. Please try again.", 500)
    log.info("analyze decision=%s bytes=%d ms=%d", result["decision"], len(data), (time.time() - t0) * 1000)

    decision = result["decision"]
    if decision == "non_food":
        return error("NON_FOOD_IMAGE", MSG_NON_FOOD, 422)
    if decision == "unsupported_food":
        return error("UNSUPPORTED_FOOD", MSG_UNSUPPORTED, 422)
    if decision == "uncertain":
        return error("LOW_CONFIDENCE", MSG_LOW_CONF, 422)

    top = result["top_k"][0]
    nut = nutrition.lookup(top["food"], serving_grams)
    warnings = [] if result["calibrated"] else ["non_food_rejection_not_calibrated"]
    return {"success": True,
            "prediction": {"food": top["food"], "class_id": top["class_id"], "confidence": round(result["confidence"], 4),
                           "margin": round(result["margin"], 4), "top_k": result["top_k"]},
            "nutrition": nut,
            "nutrition_error": None if nut else (nutrition.error or "No nutrition reference record for this food."),
            "notes": ["Confidence is the model's score, not a guarantee that the label is correct.",
                      "SmartPlateAI recognises 24 food types only."],
            "warnings": warnings}

"""Reference nutrition lookup from the project's food CSV. Missing nutrients stay None (never 0)."""
from __future__ import annotations

import json
import math
import os
import re
import threading
from pathlib import Path

import pandas as pd

BACKEND_DIR = Path(__file__).resolve().parents[2]
UNITS = {"kcal": "kcal", "mg": "mg", "ug": "Âµg", "g": "g"}
SUFFIX = re.compile(r"_(kcal|mg|ug|g)$")
BASIS = "per 100 g (USDA FoodData Central reference record)"
NOTE = ("Reference values for this type of food, not measurements of the photographed item. Real nutrition depends on portion size, "
        "ripeness and preparation.")


class NutritionReference:
    def __init__(self, csv_path: str | Path | None = None) -> None:
        self.csv_path = Path(csv_path) if csv_path else None
        self._records: dict[str, dict] | None = None
        self._lock = threading.Lock()
        self.error: str | None = None

    def _resolve(self) -> Path | None:
        if self.csv_path:
            return self.csv_path
        if os.environ.get("FOOD_NUTRITION_CSV"):
            return Path(os.environ["FOOD_NUTRITION_CSV"])
        paths = BACKEND_DIR / "reports" / "dataset_paths.json"
        if paths.is_file():
            p = json.loads(paths.read_text(encoding="utf-8")).get("food_csv")
            return Path(p) if p else None
        return None

    def load(self) -> bool:
        with self._lock:
            if self._records is not None:
                return True
            path = self._resolve()
            if path is None or not path.is_file():
                self.error = "Nutrition reference data was not found."
                return False
            header = pd.read_csv(path, nrows=0).columns
            nutrients = [c for c in header if SUFFIX.search(c)]
            df = pd.read_csv(path, usecols=["food", "fdc_id", *nutrients]).drop_duplicates("food")
            self._records = {}
            for _, row in df.iterrows():
                self._records[str(row["food"])] = {"fdc_id": None if pd.isna(row["fdc_id"]) else int(row["fdc_id"]),
                                                   "values": {c: (None if pd.isna(row[c]) else float(row[c])) for c in nutrients}}
            return True

    def lookup(self, food: str, serving_grams: float | None = None) -> dict | None:
        if not self.load() or food not in (self._records or {}):
            return None
        rec = self._records[food]
        items, missing = [], []
        for key, val in rec["values"].items():
            unit = UNITS[SUFFIX.search(key).group(1)]
            label = SUFFIX.sub("", key).replace("_", " ").capitalize()
            per_serving = None if (val is None or serving_grams is None) else round(val * serving_grams / 100.0, 3)
            if val is None:
                missing.append(label)
            items.append({"key": key, "label": label, "unit": unit, "per_100g": val, "per_serving": per_serving})
        return {"basis": BASIS, "source_fdc_id": rec["fdc_id"], "is_estimate": True, "note": NOTE, "serving_grams": serving_grams,
                "serving_note": ("Scaled linearly from the per-100 g values using the serving size you entered." if serving_grams else
                                 "No serving size was given, so only per-100 g values are shown."),
                "nutrients": items, "not_available": missing}

export type TopK = { class_id: number; food: string; score: number }
export type Nutrient = { key: string; label: string; unit: string; per_100g: number | null; per_serving: number | null }
export type Nutrition = {
  basis: string
  source_fdc_id: number | null
  is_estimate: boolean
  note: string
  serving_grams: number | null
  serving_note: string
  nutrients: Nutrient[]
  not_available: string[]
}
export type FoodSuccess = {
  success: true
  prediction: { food: string; class_id: number; confidence: number; margin: number; top_k: TopK[] }
  nutrition: Nutrition | null
  nutrition_error: string | null
  notes: string[]
  warnings: string[]
}
export type ApiError = { code: string; message: string }
export type AnalyzeResult = { ok: true; data: FoodSuccess } | { ok: false; error: ApiError }

const API_BASE = (import.meta.env.VITE_API_BASE_URL as string | undefined) ?? 'http://127.0.0.1:8000'

/** Never turns an error into a success: only an HTTP 2xx body with success === true is accepted. */
export async function analyzeFood(file: File, servingGrams?: number, signal?: AbortSignal): Promise<AnalyzeResult> {
  const form = new FormData()
  form.append('file', file)
  if (servingGrams !== undefined) form.append('serving_grams', String(servingGrams))

  let res: Response
  try {
    res = await fetch(`${API_BASE}/api/analyze/food`, { method: 'POST', body: form, signal })
  } catch (e) {
    if (e instanceof DOMException && e.name === 'AbortError') throw e
    return {
      ok: false,
      error: { code: 'BACKEND_UNAVAILABLE', message: 'Cannot reach the SmartPlateAI server. Make sure the backend is running and try again.' },
    }
  }

  let body: Record<string, unknown> | null = null
  try {
    body = (await res.json()) as Record<string, unknown>
  } catch {
    body = null
  }
  if (res.ok && body && body.success === true && typeof body.prediction === 'object' && body.prediction !== null) {
    return { ok: true, data: body as unknown as FoodSuccess }
  }
  if (body && body.success === false && typeof body.error_code === 'string' && typeof body.message === 'string') {
    return { ok: false, error: { code: body.error_code, message: body.message } }
  }
  return { ok: false, error: { code: 'UNEXPECTED_RESPONSE', message: `The server returned an unexpected response (HTTP ${res.status}). Please try again.` } }
}

export default function FoodAnalyzer() {
  const [file, setFile] = useState<File | null>(null)
  const [preview, setPreview] = useState<string | null>(null)
  const [serving, setServing] = useState('')
  const [loading, setLoading] = useState(false)
  const [result, setResult] = useState<AnalyzeResult | null>(null)
  const [localError, setLocalError] = useState<string | null>(null)
  const abortRef = useRef<AbortController | null>(null)

  const reset = () => {
    abortRef.current?.abort()
    if (preview) URL.revokeObjectURL(preview)
    setFile(null)
    setPreview(null)
    setResult(null)
    setLocalError(null)
    setServing('')
    setLoading(false)
  }

  const onPick = (e: ChangeEvent<HTMLInputElement>) => {
    const f = e.target.files?.[0]
    e.target.value = ''
    if (!f) return
    setResult(null)
    setLocalError(null)
    if (!f.type.startsWith('image/')) return setLocalError('Please choose an image file (JPG, PNG or WebP).')
    if (f.size > MAX_MB * 1024 * 1024) return setLocalError(`The image is larger than ${MAX_MB} MB. Please upload a smaller photo.`)
    if (preview) URL.revokeObjectURL(preview)
    setFile(f)
    setPreview(URL.createObjectURL(f))
  }

  const analyse = async () => {
    if (!file) return
    const grams = serving.trim() === '' ? undefined : Number(serving)
    if (grams !== undefined && (!Number.isFinite(grams) || grams < 1 || grams > 2000)) {
      return setLocalError('Serving size must be between 1 and 2000 grams.')
    }
    abortRef.current?.abort()
    const ctrl = new AbortController()
    abortRef.current = ctrl
    setLoading(true)
    setResult(null)
    setLocalError(null)
    try {
      setResult(await analyzeFood(file, grams, ctrl.signal))
    } catch {
      /* request was cancelled */
    } finally {
      if (abortRef.current === ctrl) setLoading(false)
    }
  }

  return (

            </label>
          ) : (
            <div className="grid gap-6 sm:grid-cols-2">
              <img src={preview} alt="Your uploaded food" className="aspect-square w-full rounded-2xl object-cover" />
              <div className="flex flex-col gap-4">
                <div>
                  <label htmlFor="serving" className="mb-1.5 block text-sm font-semibold">Serving size in grams (optional)</label>
                  <input
                    id="serving"

                    value={serving}
                    onChange={(e) => setServing(e.target.value)}

function ScrollManager() {
  const { pathname, hash } = useLocation()
  useEffect(() => {
    if (hash) {
      const el = document.getElementById(hash.slice(1))
      if (el) {
        el.scrollIntoView({ behavior: 'smooth' })
        return
      }
    }
    window.scrollTo({ top: 0 })
  }, [pathname, hash])
  return null
}

  )
